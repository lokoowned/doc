console.log('🚀 Script de accordion carregado!');

// Aguarda múltiplos eventos para garantir que o DOM está pronto
function initAccordion() {
  console.log('🔄 Inicializando accordion...');
  console.log('📊 Estado do DOM:', document.readyState);
  
  const allDetails = document.querySelectorAll('details');
  console.log(`✅ Encontrados ${allDetails.length} elementos <details>`);
  
  if (allDetails.length === 0) {
    console.warn('⚠️ Nenhum elemento <details> encontrado. Tentando novamente em 500ms...');
    setTimeout(initAccordion, 500);
    return;
  }
  
  // Remove eventos antigos se já foram adicionados
  allDetails.forEach(details => {
    const newDetails = details.cloneNode(true);
    details.parentNode.replaceChild(newDetails, details);
  });
  
  // Busca novamente após a clonagem
  const freshDetails = document.querySelectorAll('details');
  
  freshDetails.forEach((details, index) => {
    console.log(`🔗 Adicionando listener ao details #${index + 1}`);
    
    details.addEventListener('toggle', function(e) {
      console.log(`🎯 Toggle detectado no details #${index + 1}, estado: ${this.open ? 'ABERTO' : 'FECHADO'}`);
      
      if (this.open) {
        console.log('🔒 Fechando outros details...');
        
        // Fecha todos os outros details
        freshDetails.forEach((other, otherIndex) => {
          if (other !== this && other.open) {
            console.log(`   ↳ Fechando details #${otherIndex + 1}`);
            other.open = false;
          }
        });
        
        // Adiciona classe ao body
        document.body.classList.add('has-open-details');
        console.log('✨ Classe "has-open-details" adicionada ao body');
        
        // Scroll suave
        setTimeout(() => {
          this.scrollIntoView({ 
            behavior: 'smooth', 
            block: 'center' 
          });
        }, 100);
      } else {
        // Verifica se ainda há algum details aberto
        const hasOpenDetails = Array.from(freshDetails).some(d => d.open);
        if (!hasOpenDetails) {
          document.body.classList.remove('has-open-details');
          console.log('🌙 Classe "has-open-details" removida do body');
        }
      }
    });
  });
  
  console.log('✅ Accordion inicializado com sucesso!');
}

// Tenta inicializar em múltiplos momentos
if (document.readyState === 'loading') {
  console.log('⏳ DOM ainda carregando, aguardando DOMContentLoaded...');
  document.addEventListener('DOMContentLoaded', initAccordion);
} else {
  console.log('✅ DOM já carregado, inicializando imediatamente...');
  initAccordion();
}

// Fallback: tenta novamente após 1 segundo
setTimeout(() => {
  console.log('🔄 Tentativa de inicialização após 1 segundo (fallback)');
  initAccordion();
}, 1000);

// Observa mudanças no DOM
const observer = new MutationObserver((mutations) => {
  const hasNewDetails = mutations.some(mutation => 
    Array.from(mutation.addedNodes).some(node => 
      node.nodeName === 'DETAILS' || 
      (node.querySelectorAll && node.querySelectorAll('details').length > 0)
    )
  );
  
  if (hasNewDetails) {
    console.log('🔄 Novos elementos <details> detectados, reinicializando...');
    initAccordion();
  }
});

// Inicia observação quando o body estiver disponível
if (document.body) {
  observer.observe(document.body, { childList: true, subtree: true });
} else {
  setTimeout(() => {
    if (document.body) {
      observer.observe(document.body, { childList: true, subtree: true });
    }
  }, 100);
}
</script>
